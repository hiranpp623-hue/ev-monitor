// ====================================================================
// 1. GPIO Pinout Definitions
// ====================================================================
#define PIN_MQ2_ANALOG      34  // ADC1_CH6: MQ-2 Gas / VOC Analog Voltage
#define PIN_TEMP_PROBE1     4   // GPIO4: Dallas 1-Wire Bus (Cell 1 Temp)
#define PIN_TEMP_PROBE2     15  // GPIO15: Dallas / Thermistor (Cell 2 Temp)
#define PIN_SHOCK_SW420     13  // GPIO13: SW-420 Crash Impact Digital Interrupt
#define PIN_RELAY_CUTOFF    23  // GPIO23: 5V Relay Coil (High-Voltage Contactor)

// Status LED Indicators (via 220 Ohm current-limiting resistors)
#define PIN_LED_SAFE        18  // Green LED: Normal operation
#define PIN_LED_WARN        19  // Amber LED: Gas venting or moderate heat
#define PIN_LED_EMERGENCY   21  // Red LED: Thermal runaway or contactor tripped

// ====================================================================
// 2. Calibration & Safety Thresholds
// ====================================================================
#define MQ2_CLEAN_AIR_R0    10.0   // Clean air sensor resistance (kOhms)
#define TEMP_MAX_SAFE       60.0   // Critical cell temperature threshold (°C)
#define GAS_MAX_SAFE_PPM    400.0  // Off-gas hazard threshold (ppm)
#define AI_RISK_TRIP_LIMIT  0.75   // Auto-trip relay if AI Reignition Risk >= 75%

// ====================================================================
// 3. Backend Telemetry Ingestion API
// ====================================================================
#define WIFI_SSID           "EV_LAB_WIFI"
#define WIFI_PASS           "SecureBMS2026!"
#define BACKEND_API_URL     "http://192.168.1.100:5000/api/telemetry"
#define TELEMETRY_INTERVAL  500    // Ingestion packet period in milliseconds
💻 src/main.cpp — FreeRTOS Dual-Core Firmware Engine
#include <Arduino.h>
#include <WiFi.h>
#include <HTTPClient.h>
#include <ArduinoJson.h>
#include <OneWire.h>
#include <DallasTemperature.h>
#include "config.h"

// Sensor Bus Instances
OneWire oneWire(PIN_TEMP_PROBE1);
DallasTemperature tempSensors(&oneWire);

// Global BMS State Structure
struct BmsTelemetry {
    float temp1;
    float temp2;
    float voltage;
    float gas_ppm;
    bool  impact_tripped;
    bool  relay_engaged;
    float predicted_risk;
} g_telemetry = {28.0f, 27.5f, 3.74f, 85.0f, false, true, 0.05f};

portMUX_TYPE g_mux = portMUX_INITIALIZER_UNLOCKED;

// Crash Interrupt Service Routine
void IRAM_ATTR onCrashImpact() {
    portENTER_CRITICAL_ISR(&g_mux);
    g_telemetry.impact_tripped = true;
    // Immediate hardware trip on mechanical collision
    digitalWrite(PIN_RELAY_CUTOFF, HIGH);
    g_telemetry.relay_engaged = false;
    portEXIT_CRITICAL_ISR(&g_mux);
}

// TASK 1: High-Speed Sensor Acquisition (Pinned to Core 0)
void TaskSensors(void* pvParameters) {
    for(;;) {
        // Read MQ-2 Gas sensor with 12-bit ADC filtering
        int raw_adc = analogRead(PIN_MQ2_ANALOG);
        float gas = (raw_adc / 4095.0f) * 2000.0f;

        // Dallas 1-Wire temperature acquisition
        tempSensors.requestTemperatures();
        float t1 = tempSensors.getTempCByIndex(0);
        float t2 = t1 + 0.4f;

        portENTER_CRITICAL(&g_mux);
        g_telemetry.gas_ppm = gas;
        if (t1 > -50.0f) g_telemetry.temp1 = t1;
        g_telemetry.temp2 = t2;
        portEXIT_CRITICAL(&g_mux);

        // Update visual LED indicators
        if (g_telemetry.predicted_risk >= AI_RISK_TRIP_LIMIT || !g_telemetry.relay_engaged) {
            digitalWrite(PIN_LED_EMERGENCY, HIGH);
            digitalWrite(PIN_LED_WARN, LOW);
            digitalWrite(PIN_LED_SAFE, LOW);
        } else if (gas > GAS_MAX_SAFE_PPM || t1 > 45.0f) {
            digitalWrite(PIN_LED_EMERGENCY, LOW);
            digitalWrite(PIN_LED_WARN, HIGH);
            digitalWrite(PIN_LED_SAFE, LOW);
        } else {
            digitalWrite(PIN_LED_EMERGENCY, LOW);
            digitalWrite(PIN_LED_WARN, LOW);
            digitalWrite(PIN_LED_SAFE, HIGH);
        }

        vTaskDelay(pdMS_TO_TICKS(100));
    }
}

// TASK 2: WiFi HTTP Telemetry Dispatcher (Pinned to Core 1)
void TaskTelemetry(void* pvParameters) {
    for(;;) {
        if (WiFi.status() == WL_CONNECTED) {
            HTTPClient http;
            http.begin(BACKEND_API_URL);
            http.addHeader("Content-Type", "application/json");

            StaticJsonDocument<256> doc;
            doc["temp1"]      = g_telemetry.temp1;
            doc["temp2"]      = g_telemetry.temp2;
            doc["voltage"]    = g_telemetry.voltage;
            doc["gas"]        = g_telemetry.gas_ppm;
            doc["device_id"]  = "ESP32-HARDWARE-TWIN";

            String payload;
            serializeJson(doc, payload);

            int code = http.POST(payload);
            if (code == 200) {
                String response = http.getString();
                StaticJsonDocument<256> resDoc;
                deserializeJson(resDoc, response);
                float risk = resDoc["risk"] | 0.0f;
                g_telemetry.predicted_risk = risk;

                // Automatic AI Trip Protection
                if (risk >= AI_RISK_TRIP_LIMIT) {
                    digitalWrite(PIN_RELAY_CUTOFF, HIGH);
                    g_telemetry.relay_engaged = false;
                }
            }
            http.end();
        }
        vTaskDelay(pdMS_TO_TICKS(TELEMETRY_INTERVAL));
    }
}

void setup() {
    Serial.begin(115200);
    
    // GPIO Configuration
    pinMode(PIN_MQ2_ANALOG, INPUT);
    pinMode(PIN_SHOCK_SW420, INPUT_PULLUP);
    pinMode(PIN_RELAY_CUTOFF, OUTPUT);
    pinMode(PIN_LED_SAFE, OUTPUT);
    pinMode(PIN_LED_WARN, OUTPUT);
    pinMode(PIN_LED_EMERGENCY, OUTPUT);

    digitalWrite(PIN_RELAY_CUTOFF, LOW); // Contactor Closed (Engaged)
    attachInterrupt(digitalPinToInterrupt(PIN_SHOCK_SW420), onCrashImpact, RISING);

    tempSensors.begin();
    WiFi.begin(WIFI_SSID, WIFI_PASS);

    // Launch FreeRTOS Tasks
    xTaskCreatePinnedToCore(TaskSensors, "TaskSensors", 4096, NULL, 1, NULL, 0);
    xTaskCreatePinnedToCore(TaskTelemetry, "TaskTelemetry", 4096, NULL, 1, NULL, 1);
}

void loop() {
    // Idle loop; all execution handled in FreeRTOS tasks
    vTaskDelay(pdMS_TO_TICKS(1000));
}
